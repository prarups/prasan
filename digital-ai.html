barcode-generator.py 
import requests

# Prasan Global Barcode & Sticker Printing API
url = "https://api.prasanglobal.in/v1/barcode/generate"
payload = {
    "product_sku": "PRASAN-GARMENT-1002",
    "barcode_type": "EAN-13",
    "sticker_layout": "50mmx25mm_thermal"
}

response = requests.post(url, json=payload)
print("Barcode Generated:", response.json()["barcode_url"])
